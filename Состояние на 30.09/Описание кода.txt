Этот скетч превращает ESP8266 в Wi-Fi пульт для робота.
ESP8266 подключается к существующей Wi-Fi сети как клиент, 
получает IP-адрес и запускает веб-сервер на порту 80. 
В браузере открывается страница с кнопками: вперёд, назад, влево, вправо и стоп (при 
заклинивании механизма, на всякий случай).
При нажатии кнопки браузер отправляет GET-запрос вида:
/cmd?action=F/B/L/R/S 

Сервер принимает команду и вызывает нужную функцию,
которая подаёт логические уровни на входы драйвера L9110.
Драйвер управляет двумя моторами: вперёд/назад — оба мотора крутятся в одну сторону, 
влево/вправо — моторы вращаются в противоположные стороны,
 стоп — все входы выключены.

 В loop() ESP8266 постоянно обрабатывает HTTP-запросы.

#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>

const char* ssid = "LOGIN";
const char* password = "PASSWORD";

#define MOTOR_A_IA D1
#define MOTOR_A_IB D2
#define MOTOR_B_IA D5
#define MOTOR_B_IB D6

const int MOTOR_SPEED = 50;
ESP8266WebServer server(80);

void stopMotors() {
  digitalWrite(MOTOR_A_IA, LOW);
  digitalWrite(MOTOR_A_IB, LOW);
  digitalWrite(MOTOR_B_IA, LOW);
  digitalWrite(MOTOR_B_IB, LOW);
}

void moveForward() {
  digitalWrite(MOTOR_A_IA, MOTOR_SPEED);
  digitalWrite(MOTOR_A_IB, LOW);
  digitalWrite(MOTOR_B_IA, MOTOR_SPEED);
  digitalWrite(MOTOR_B_IB, LOW);
}

void moveBackward() {
  digitalWrite(MOTOR_A_IA, LOW);
  digitalWrite(MOTOR_A_IB, MOTOR_SPEED);
  digitalWrite(MOTOR_B_IA, LOW);
  digitalWrite(MOTOR_B_IB, MOTOR_SPEED);
}

void turnLeft() {
  digitalWrite(MOTOR_A_IA, MOTOR_SPEED);
  digitalWrite(MOTOR_A_IB, LOW);
  digitalWrite(MOTOR_B_IA, LOW);
  digitalWrite(MOTOR_B_IB, MOTOR_SPEED);
}

void turnRight() {
  digitalWrite(MOTOR_A_IA, LOW);
  digitalWrite(MOTOR_A_IB, MOTOR_SPEED);
  digitalWrite(MOTOR_B_IA, MOTOR_SPEED);
  digitalWrite(MOTOR_B_IB, LOW);
}

void handleRoot() {
  server.send_P(200, "text/html", index_html);
}

void handleCommand() {
  String action = server.arg("action");
  Serial.println("Команда: " + action);

  if      (action == "F") moveForward();
  else if (action == "B") moveBackward();
  else if (action == "L") turnLeft();
  else if (action == "R") turnRight();
  else                    stopMotors();

  server.send(200, "text/plain", "OK");
}

void handleNotFound() {
  server.send(404, "text/plain", "Not Found");
}

void setup() {
  Serial.begin(115200);
  delay(100);

  pinMode(MOTOR_A_IA, OUTPUT);
  pinMode(MOTOR_A_IB, OUTPUT);
  pinMode(MOTOR_B_IA, OUTPUT);
  pinMode(MOTOR_B_IB, OUTPUT);
  stopMotors();

  Serial.print("Подключение к Wi-Fi: ");
  Serial.println(ssid);
  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nWi-Fi подключен!");
  Serial.print("IP-адрес: ");
  Serial.println(WiFi.localIP());

  server.on("/", handleRoot);
  server.on("/cmd", handleCommand);
  server.onNotFound(handleNotFound);
  server.begin();
  Serial.println("HTTP-сервер запущен. Откройте браузер и перейдите по IP-адресу выше.");
}

void loop() {
  server.handleClient();
}
